    const scrollTopBtn = document.getElementById('scrollTopBtn');

    function throttle(func, limit) {
        let inThrottle;
        return function() {
            if (!inThrottle) {
                func.apply(this, arguments);
                inThrottle = true;
                setTimeout(() => inThrottle = false, limit);
            }
        };
    }

    window.addEventListener('scroll', throttle(function() {
        scrollTop();
    }, 100));

        window.scrollTo({

            left: 0,
            behavior: 'smooth'
        });
    });

        } else {
